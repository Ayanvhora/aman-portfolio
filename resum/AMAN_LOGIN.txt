import { db } from "./firebase.js";

import {
    collection,
    getDocs
} from "https://www.gstatic.com/firebasejs/12.16.0/firebase-firestore.js";

const projectsGrid = document.getElementById("projectsGrid");

async function loadFirebaseProjects() {
    if (!projectsGrid) {
        console.error("Projects grid not found. Check projects.html.");
        return;
    }

    try {
        const snapshot = await getDocs(collection(db, "projects"));

        console.log("Firebase project documents:", snapshot.size);

        const projects = snapshot.docs.map((document) => ({
            id: document.id,
            ...document.data()
        }));

        projects.sort((a, b) => (a.order ?? 9999) - (b.order ?? 9999));

        projects.forEach((project) => {
            if (!project.title) {
                console.warn("Project has no title:", project.id);
                return;
            }

            const card = document.createElement("article");
            card.className = "project-card reveal";

            // Project image
            const imageUrl = project.imageUrl || project.imageURL;

            if (imageUrl) {
                const image = document.createElement("img");
                image.src = imageUrl;
                image.alt = project.title;
                image.loading = "lazy";

                image.onerror = () => {
                    image.style.display = "none";
                };

                card.appendChild(image);
            }

            const content = document.createElement("div");
            content.className = "project-content";

            // Title
            const title = document.createElement("h2");
            title.textContent = project.title;
            content.appendChild(title);

            // Description
            const description = document.createElement("p");
            description.textContent =
                project.description ||
                project.summary ||
                "Explore this project and its key details.";

            content.appendChild(description);

            // Technologies
            if (project.technologies) {
                const technologies = document.createElement("p");
                technologies.className = "project-technologies";
                technologies.textContent =
                    "Technologies: " + project.technologies;

                content.appendChild(technologies);
            }

            // Project links
            if (Array.isArray(project.projectLinks)) {
                project.projectLinks.forEach((item) => {
                    if (!item.url) return;

                    const link = document.createElement("a");
                    link.href = item.url;
                    link.textContent = item.title || "Project Link";
                    link.className = "btn";
                    link.target = "_blank";
                    link.rel = "noopener noreferrer";

                    content.appendChild(link);
                });
            }

            // Uploaded files
            if (Array.isArray(project.files) && project.files.length > 0) {
                const filesContainer = document.createElement("div");
                filesContainer.className = "project-files";

                project.files.forEach((file) => {
                    if (!file.url) return;

                    const link = document.createElement("a");
                    link.href = file.url;
                    link.textContent = "Open " + (file.name || "Project File");
                    link.target = "_blank";
                    link.rel = "noopener noreferrer";
                    link.className = "btn";

                    filesContainer.appendChild(link);
                });

                if (filesContainer.children.length > 0) {
                    content.appendChild(filesContainer);
                }
            }

           // View Project button
            const viewButton = document.createElement("a");

            viewButton.className = "btn";
            viewButton.textContent = "View Project";
            viewButton.href = "#";

            viewButton.addEventListener("click", (event) => {
                event.preventDefault();
                showProjectDetails(project);
            });

            content.appendChild(viewButton);
            card.appendChild(content);

            projectsGrid.appendChild(card);
        });

        // Reveal animation
        projectsGrid.querySelectorAll(".reveal").forEach((element) => {
            const top = element.getBoundingClientRect().top;

            if (top < window.innerHeight - 50) {
                element.classList.add("active");
            }
        });

    } catch (error) {
        console.error("Failed to load Firebase projects:", error);
    }
}

loadFirebaseProjects();

// FULL-SCREEN PROJECT DETAILS WINDOW
// ======================================

function showProjectDetails(project) {

    // Remove an existing details window
    const oldModal = document.getElementById("projectDetailsModal");

    if (oldModal) {
        oldModal.remove();
    }

    // Main modal
    const modal = document.createElement("div");
    modal.id = "projectDetailsModal";
    modal.className = "project-details-modal";

    // Header
    const header = document.createElement("div");
    header.className = "project-details-header";

    const heading = document.createElement("h2");
    heading.textContent = project.title || "Project Details";

    const closeButton = document.createElement("button");
    closeButton.className = "project-details-close";
    closeButton.textContent = "×";
    closeButton.setAttribute("aria-label", "Close project details");

    closeButton.addEventListener("click", () => {
        modal.remove();
        document.body.style.overflow = "";
    });

    header.appendChild(heading);
    header.appendChild(closeButton);

    // Main content
    const body = document.createElement("div");
    body.className = "project-details-body";

    // Project image
    const imageUrl = project.imageUrl || project.imageURL;

    if (imageUrl) {
        const image = document.createElement("img");
        image.className = "project-details-image";
        image.src = imageUrl;
        image.alt = project.title || "Project image";

        body.appendChild(image);
    }

    // Project summary
    const summarySection = document.createElement("section");
    summarySection.className = "project-details-section";

    const summaryHeading = document.createElement("h3");
    summaryHeading.textContent = "Project Overview";

    const summaryText = document.createElement("p");
    summaryText.textContent =
        project.summary ||
        project.description ||
        "No project summary has been added yet.";

    summarySection.appendChild(summaryHeading);
    summarySection.appendChild(summaryText);

    body.appendChild(summarySection);

    // Technologies
    const technologySection = document.createElement("section");
    technologySection.className = "project-details-section";

    const technologyHeading = document.createElement("h3");
    technologyHeading.textContent = "Technologies Used";

    const technologyText = document.createElement("p");
    technologyText.textContent =
        project.technologies || "Technologies not specified.";

    technologySection.appendChild(technologyHeading);
    technologySection.appendChild(technologyText);

    body.appendChild(technologySection);

    // Uploaded project files
    const filesSection = document.createElement("section");
    filesSection.className = "project-details-section";

    const filesHeading = document.createElement("h3");
    filesHeading.textContent = "Project Files";

    filesSection.appendChild(filesHeading);

    const files = Array.isArray(project.files)
        ? project.files
        : [];

    if (files.length === 0) {
        const noFiles = document.createElement("p");
        noFiles.textContent = "No project files have been uploaded.";
        filesSection.appendChild(noFiles);
    } else {

        const filesGrid = document.createElement("div");
        filesGrid.className = "project-details-files-grid";

        files.forEach((file) => {

            if (!file.url) return;

            const fileCard = document.createElement("div");
            fileCard.className = "project-details-file-card";

            const fileName = document.createElement("h4");
            fileName.textContent = file.name || "Project File";

            const fileType = document.createElement("p");

            const extension = (file.name || "")
                .split(".")
                .pop()
                .toUpperCase();

            fileType.textContent =
                extension && extension !== (file.name || "").toUpperCase()
                    ? extension + " File"
                    : "Project Resource";

            const fileActions = document.createElement("div");
            fileActions.className = "project-details-file-actions";

            // Open or preview project file
            
            // Smart Project File Preview Buttons
            const openLink = document.createElement("a");

            const actualFileName = (file.name || "").trim().toLowerCase();
            const fileExt = extension.toLowerCase();

            const fileCategory = (
                file.category ||
                file.fileType ||
                ""
            ).toLowerCase().trim();

            let buttonText = "Open Project File";
            let previewUrl = file.url;

            // Detect file type
            if (
                fileExt === "xlsx" ||
                fileExt === "xls" ||
                fileExt === "xlsm"
            ) {
                if (
                    fileCategory === "dashboard" ||
                    fileCategory === "excel-dashboard" ||
                    actualFileName.includes("dashboard")
                ) {
                    buttonText = "Open Excel File";

                    previewUrl =
                        "https://view.officeapps.live.com/op/view.aspx?src=" +
                        encodeURIComponent(file.url);
                } else {
                    buttonText = "Open Excel File";
                }
            }
            else if (fileExt === "pdf") {
                buttonText = "View PDF Report";
            }
            else if (fileExt === "csv") {
                buttonText = "View Dataset";
            }
            else if (fileExt === "py") {
                buttonText = "View Python Code";
            }
            else if (fileExt === "ipynb") {
                buttonText = "View Notebook";
            }
            else if (fileExt === "html" || fileExt === "htm") {
                buttonText = "View Web Project";
            }

            openLink.href = previewUrl;
            openLink.textContent = buttonText;
            openLink.target = "_blank";
            openLink.rel = "noopener noreferrer";
            openLink.className = "project-details-link";

            fileActions.appendChild(openLink);



            const downloadLink = document.createElement("a");
            downloadLink.href = file.url;
            downloadLink.textContent = "Download";
            downloadLink.target = "_blank";
            downloadLink.rel = "noopener noreferrer";
            downloadLink.className = "project-details-link secondary";

            fileActions.appendChild(downloadLink);

            fileCard.appendChild(fileName);
            fileCard.appendChild(fileType);
            fileCard.appendChild(fileActions);

            filesGrid.appendChild(fileCard);
        });

        filesSection.appendChild(filesGrid);
    }

    body.appendChild(filesSection);

    // Project links
    const linksSection = document.createElement("section");
    linksSection.className = "project-details-section";

    const linksHeading = document.createElement("h3");
    linksHeading.textContent = "Project Links";

    linksSection.appendChild(linksHeading);

    const projectLinks = Array.isArray(project.projectLinks)
        ? project.projectLinks
        : [];

    if (projectLinks.length === 0) {
        const noLinks = document.createElement("p");
        noLinks.textContent = "No additional project links have been added.";
        linksSection.appendChild(noLinks);
    } else {

        const linksGrid = document.createElement("div");
        linksGrid.className = "project-details-links-grid";

        projectLinks.forEach((item) => {

            if (!item.url) return;

            const link = document.createElement("a");
            link.href = item.url;
            link.textContent = item.title || "Open Project Link";
            link.target = "_blank";
            link.rel = "noopener noreferrer";
            link.className = "project-details-link";

            linksGrid.appendChild(link);
        });

        linksSection.appendChild(linksGrid);
    }

    body.appendChild(linksSection);

    // Footer
    const footer = document.createElement("div");
    footer.className = "project-details-footer";

    const footerText = document.createElement("p");
    footerText.textContent = "Project Portfolio";

    footer.appendChild(footerText);

    // Assemble modal
    modal.appendChild(header);
    modal.appendChild(body);
    modal.appendChild(footer);

    document.body.appendChild(modal);

    // Prevent background scrolling
    document.body.style.overflow = "hidden";

    // Close when clicking outside the content
    modal.addEventListener("click", (event) => {
        if (event.target === modal) {
            modal.remove();
            document.body.style.overflow = "";
        }
    });

    // Close with Escape key
    function handleEscape(event) {
        if (event.key === "Escape") {
            modal.remove();
            document.body.style.overflow = "";
            document.removeEventListener("keydown", handleEscape);
        }
    }

    document.addEventListener("keydown", handleEscape);

    // Start at the top of the details window
    body.scrollTop = 0;
}
window.showProjectDetails = showProjectDetails;
